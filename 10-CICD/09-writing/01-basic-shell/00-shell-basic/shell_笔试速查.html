条件

空格铁律：[ 后、] 前、运算符两侧都要有空格；变量加引号
[ "$a" = 1 ]               # 对
[$a=1]                     # 错
[[ ... ]]                  # bash 扩展，更安全，推荐

字符串:  =  !=  -z空  -n非空
数值:    -eq -ne -gt -ge -lt -le     （[ ]里不要用 > <，会被当重定向）
文件:    -f文件 -d目录 -x可执行 -r可读 -e存在

if [ "$n" -gt 100 ]; then
  echo big
elif [ "$n" -eq 100 ]; then
  echo equal
else
  echo small
fi

[ -f a ] && [ -r a ]       # 与
[ -f a ] || [ -f b ]       # 或
[[ -f a && -r a ]]         # [[ ]] 里可用 && ||

完整模板：跑测试汇总

#!/usr/bin/env bash
set -euo pipefail
TEST_DIR="${1:-./tests}"
REPORT="report_$(date +%Y%m%d_%H%M%S).txt"
pass=0; fail=0
run_one() {
  local name="$1"
  if bash "$name"; then
    echo "[PASS] $name" | tee -a "$REPORT"; pass=$((pass+1))
  else
    echo "[FAIL] $name" | tee -a "$REPORT"; fail=$((fail+1))
  fi
}
[ -d "$TEST_DIR" ] || { echo "no dir" >&2; exit 1; }
for t in "$TEST_DIR"/*.sh; do
  [ -e "$t" ] || continue
  run_one "$t"
done
echo "pass=$pass fail=$fail" | tee -a "$REPORT"
[ "$fail" -eq 0 ]          # 有失败则非0，接 CI
